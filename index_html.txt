    <script>
        // Generate dataset markers spanning the typical crop duration lifecycle
        const labels = [];
        const heightData = [];
        const derivativeData = [];

        function getHeight(x) {
            return (-0.0006 * Math.pow(x, 3)) + (0.092 * Math.pow(x, 2)) - (1.2 * x) + 10;
        }

        function getDerivative(x) {
            return (-0.0018 * Math.pow(x, 2)) + (0.184 * x) - 1.2;
        }

        // Generate data at 5-day intervals across the 100 day crop span
        for (let x = 5; x <= 100; x += 5) {
            labels.push(x);
            heightData.push(parseFloat(getHeight(x).toFixed(1)));
            derivativeData.push(parseFloat(getDerivative(x).toFixed(2)));
        }

        function updateInspector(day, hVal, dVal) {
            document.getElementById('welcomePrompt').classList.add('hidden');
            document.getElementById('inspectorContent').classList.remove('hidden');
            
            document.getElementById('outAge').innerText = "Day " + day;
            document.getElementById('outHeight').innerText = hVal.toFixed(1) + " cm";
            document.getElementById('outDerivative').innerText = dVal.toFixed(2) + " cm/day";
            
            const panel = document.getElementById('inspectorPanel');
            const expText = document.getElementById('outExplanation');
            
            // Contextual rules based on physiological phases of corn development
            if (day <= 30) {
                panel.className = "bg-stone-900 rounded-2xl p-6 text-white shadow-xl transition-all duration-300 border-2 border-amber-500";
                expText.innerHTML = `At <strong>Day ${day}</strong>, the corn is in its early <strong>Vegetative Lag Phase</strong>. The growth rate is relatively low (+${dVal.toFixed(2)} cm/day) as energy is directed down to build the nodal root architecture below the soil line. Keep basic herbicide applications steady.`;
            } else if (day > 30 && day <= 60) {
                panel.className = "bg-stone-900 rounded-2xl p-6 text-white shadow-xl transition-all duration-300 border-2 border-emerald-500";
                expText.innerHTML = `At <strong>Day ${day}</strong>, the corn is in its peak <strong>Logarithmic Rapid Elongation Phase</strong>. Growth is aggressive at +${dVal.toFixed(2)} cm/day! The derivative shows that structural elongation is near or at maximum speed. This is the ultimate window for nitrogen side-dressing fertilization to optimize kernel potential.`;
            } else if (day > 60 && day <= 75) {
                panel.className = "bg-stone-900 rounded-2xl p-6 text-white shadow-xl transition-all duration-300 border-2 border-teal-500";
                expText.innerHTML = `At <strong>Day ${day}</strong>, the derivative is dropping (+${dVal.toFixed(2)} cm/day) even though the plant is tall. The corn has transitioned to the <strong>Tasseling and Silking reproductive phases</strong>. Vertical height growth is deliberately braking as structural resource allocation moves into kernel generation.`;
            } else {
                panel.className = "bg-stone-900 rounded-2xl p-6 text-white shadow-xl transition-all duration-300 border-2 border-stone-500";
                expText.innerHTML = `At <strong>Day ${day}</strong>, the growth speed derivative has dropped near zero (+${dVal.toFixed(2)} cm/day). The plant has achieved its absolute maximum structural height. It has reached <strong>Physiological Maturity</strong>; moisture levels are dropping in the field as kernels dry down, signaling preparation for machine mechanical harvesting.`;
            }
        }

        // Render Chart canvas
        const ctx = document.getElementById('cornChart').getContext('2d');
        const cornChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [{
                    label: 'Corn Height f(x)',
                    data: heightData,
                    borderColor: 'rgb(217, 119, 6)',
                    backgroundColor: 'rgba(217, 119, 6, 0.08)',
                    borderWidth: 3,
                    fill: true,
                    tension: 0.3,
                    pointRadius: 6,
                    pointHoverRadius: 9,
                    pointBackgroundColor: 'rgb(16, 185, 129)'
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: { legend: { display: false } },
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: 'Days Post-Planting (Lifecycle timeline)',
                            font: { weight: 'bold' }
                        }
                    },
                    y: {
                        title: {
                            display: true,
                            text: 'Standing Total Crop Height (cm)',
                            font: { weight: 'bold' }
                        },
                        min: 0,
                        max: 220
                    }
                },
                onClick: (e) => {
                    const points = cornChart.getElementsAtEventForMode(e, 'nearest', { intersect: false }, true);
                    if (points.length > 0) {
                        const idx = points[0].index;
                        updateInspector(labels[idx], heightData[idx], derivativeData[idx]);
                        cornChart.setActiveElements([{datasetIndex: 0, index: idx}]);
                        cornChart.update();
                    }
                }
            }
        });
    </script>
